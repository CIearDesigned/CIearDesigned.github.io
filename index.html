function renderMessages() { const thread = document.getElementById("messageThread"); const currentUser = getCurrentUser(); const isAdmin = !!currentUser && ( (currentUser.email && currentUser.email.toLowerCase() === ADMIN_EMAIL.toLowerCase()) || currentUser.name === ADMIN_USERNAME ); const messages = getMessages(); if (!messages || messages.length === 0) { thread.innerHTML = '
No messages yet. Start the conversation!
'; return; } let htmlContent = ''; messages.forEach((msg) => { const isFromAdmin = msg.sender === ADMIN_USERNAME; const isFromCurrentUser = msg.sender === currentUser?.name; // Admin sees all messages // Non-admin users see only their own messages + admin responses const shouldShow = isAdmin || isFromCurrentUser || isFromAdmin; if (shouldShow) { htmlContent += `
${msg.sender}
${msg.text}
${new Date(msg.time).toLocaleString()}
`; } }); if (htmlContent === '') { thread.innerHTML = '
No messages yet. Start the conversation!
'; } else { thread.innerHTML = htmlContent; } thread.scrollTop = thread.scrollHeight; }
